    function unwrap(value) {
        let result = value;

        for (let i = 0; i < 5; i++) {
            if (Array.isArray(result)) return result;
            if (!result || typeof result !== 'object') return result;

            const next =
                result.data ??
                result.Data ??
                result.result ??
                result.Result ??
                result.items ??
                result.Items;

            if (next === undefined || next === result) break;
            result = next;
        }

        return result;
    }

    async function request(path, options = {}) {
        let response;

        try {
            response = await fetch(API + path, {
                ...options,
                credentials: 'include',
                headers: {
                    Accept: 'application/json',
                    ...(options.body !== undefined
                        ? { 'Content-Type': 'application/json' }
                        : {}),
                    ...(options.headers || {})
                },
                body: options.body !== undefined
                    ? (
                        typeof options.body === 'string'
                            ? options.body
                            : JSON.stringify(options.body)
                    )
                    : undefined
            });
        } catch (error) {
            console.error('KhÃ´ng káº¿t ná»‘i Ä‘Æ°á»£c API:', API + path, error);
            throw new Error(
                `KhÃ´ng káº¿t ná»‘i Ä‘Æ°á»£c API ${API + path}. Kiá»ƒm tra API Ä‘ang cháº¡y, chá»©ng chá»‰ HTTPS vÃ  cáº¥u hÃ¬nh CORS.`
            );
        }

        const raw = await response.text();
        let data = null;

        try {
            data = raw ? JSON.parse(raw) : null;
        } catch {
            data = raw;
        }

        if (!response.ok) {
            const detail = typeof data === 'object' && data
                ? get(data, 'message', 'Message', 'title', 'Title', 'detail', 'Detail')
                : '';

            if (response.status === 401) {
                throw new Error(
                    detail || 'API tráº£ vá» 401: chÆ°a Ä‘Äƒng nháº­p hoáº·c Session Ä‘Ã£ háº¿t háº¡n.'
                );
            }

            if (response.status === 403) {
                throw new Error(
                    detail || 'API tráº£ vá» 403: Session khÃ´ng cÃ³ quyá»n Admin.'
                );
            }

            throw new Error(
                detail || `API tráº£ vá» HTTP ${response.status} táº¡i ${path}.`
            );
        }

        if (
            data &&
            typeof data === 'object' &&
            (
                data.success === false ||
                data.Success === false
            )
        ) {
            throw new Error(
                get(data, 'message', 'Message') || 'API tráº£ vá» lá»—i.'
            );
        }

        return data;
    }

    function openModal(title, body, onSave, saveText = 'LÆ°u') {

            return;
        }

        const titleElement = $('#modalTitle', modal);

            notice('Modal thiáº¿u modalTitle, modalBody hoáº·c modalSave.', 'error');
            return;
        }

        titleElement.textContent = title;
        bodyElement.innerHTML = body;
        saveButton.textContent = saveText;

        modal.hidden = false;
        modal.classList.add('open');

        saveButton.onclick = async () => {
            saveButton.disabled = true;

            try {
                await onSave();
                closeModal();
            } catch (error) {
                notice(error.message || 'KhÃ´ng thá»ƒ lÆ°u dá»¯ liá»‡u.', 'error');
            } finally {
                saveButton.disabled = false;
            }
        };
    }

    function bindActions() {

                        await openBookForm();

                    case 'book-edit': {

                            state.books,

                        );

                        break;

                        break;

                        break;

                        );

                        break;

                        break;

                        );

                        await openSimpleForm('publisher', item);
                        break;

                    case 'customer-detail':
                        await showCustomerDetail(id);
                        break;

                    case 'order-detail':

        document.getElementById('api-book-search') ||
        document.getElementById('search-book-name');

    const sort = $('#book-sort');
    const params = new URLSearchParams(window.location.search);

    // Nháº­n cáº£ hai tÃªn tham sá»‘ URL
    const urlKeyword =
        params.get('keyword') ||
        params.get('q') ||
        '';

    // Æ¯u tiÃªn tá»« khÃ³a trÃªn URL khi vá»«a chuyá»ƒn tá»« trang chá»§ sang Shop
    const keyword = (
        urlKeyword ||
        search?.value?.trim() ||
        ''
    ).toLocaleLowerCase('vi');

    // Äá»c Ä‘Ãºng mÃ£ thá»ƒ loáº¡i tá»« URL hoáº·c dropdown
    const categorySelect =
        document.getElementById('search-book-category') ||
        document.getElementById('api-book-category');

    const category =
        params.get('loai') ||
        categorySelect?.value ||
        state.category ||
        '';

    const sortType = sort ? sort.value : 'default';

    
    function getCategoryId(item) {
    const category =
        item?.maTheLoaiNavigation ??
        item?.MaTheLoaiNavigation ??
        item?.theLoai ??
        item?.TheLoai ??
        item?.theLoaiSach ??
        item?.TheLoaiSach;

    return (
        item?.maTheLoai ??
        item?.MaTheLoai ??
        item?.theLoaiId ??
        item?.TheLoaiId ??
        item?.idTheLoai ??
        item?.IdTheLoai ??
        category?.maTheLoai ??
        category?.MaTheLoai ??
        category?.theLoaiId ??
        category?.TheLoaiId ??
        category?.id ??
        category?.Id ??
        null
    );
}

    const getPrice = book =>
        Number(book.gia ?? book.Gia ?? book.price ?? 0);

    const getQuantity = book =>
        Number(book.soLuong ?? book.SoLuong ?? book.quantity ?? 0);

    const getYear = book =>
        Number(book.namXuatBan ?? book.NamXuatBan ?? book.year ?? 0);

    // 1. Lá»c sÃ¡ch
    let books = (state.books || []).filter(book => {
    const title = String(bookTitle(book) || '')
        .trim()
        .toLocaleLowerCase('vi');

    const matchesKeyword =
        !keyword || title.includes(keyword);

    const bookCategoryId = getCategoryId(book);
    const bookCategoryName = getCategoryName(book);

    const selectedCategory = (
        document.querySelector(
            '#api-book-categories [data-category="' + category + '"]'
        )?.textContent || ''
    ).trim().toLocaleLowerCase('vi');

    const matchesCategory =
        !category ||
        (
            bookCategoryId != null &&
            String(bookCategoryId) === String(category)
        ) ||
        (
            bookCategoryName !== '' &&
            selectedCategory !== '' &&
            bookCategoryName.includes(selectedCategory)
        );

    console.log({
        tenSach: bookTitle(book),
        maTheLoaiSach: bookCategoryId,
        maTheLoaiDangChon: category,
        matchesKeyword,
        matchesCategory
    });

    return matchesKeyword && matchesCategory;
});

    // 2. Sáº¯p xáº¿p
    switch (sortType) {
        case 'name-asc':
            books.sort((a, b) =>
                String(bookTitle(a)).localeCompare(
                    String(bookTitle(b)), 'vi'
                )
            );
            break;

        case 'name-desc':
            books.sort((a, b) =>
                String(bookTitle(b)).localeCompare(
                    String(bookTitle(a)), 'vi'
                )
            );
            break;

        case 'price-asc':
            books.sort((a, b) => getPrice(a) - getPrice(b));
            break;

        case 'price-desc':
            books.sort((a, b) => getPrice(b) - getPrice(a));
            break;

        case 'quantity-desc':
            books.sort((a, b) => getQuantity(b) - getQuantity(a));
            break;

        case 'year-desc':
            books.sort((a, b) => getYear(b) - getYear(a));
            break;
    }

    // 3. TÃ­nh phÃ¢n trang
    const totalBooks = books.length;
    const totalPages = Math.ceil(totalBooks / BOOKS_PER_PAGE);

    if (totalPages === 0) {
        currentBookPage = 1;
    } else {
        currentBookPage = Math.max(
            1,
            Math.min(currentBookPage, totalPages)
        );
    }

    const startIndex = (currentBookPage - 1) * BOOKS_PER_PAGE;
    const pageBooks = books.slice(
        startIndex,
        startIndex + BOOKS_PER_PAGE
    );

    // 4. Trang chá»§: giá»¯ cÃ¡ch hiá»ƒn thá»‹ 8 sÃ¡ch
    const homeEl = $('#api-books-home');

    if (homeEl) {
        const homeBooks = keyword || category
            ? books.slice(0, 8)
            : books.slice(0, 8);

        homeEl.innerHTML = homeBooks.length
            ? homeBooks.map(card).join('')
            : '<p class="text-muted">KhÃ´ng tÃ¬m tháº¥y sÃ¡ch phÃ¹ há»£p.</p>';
    }

    // 5. Hiá»ƒn thá»‹ sÃ¡ch cá»§a trang hiá»‡n táº¡i trong shop.html
    const gridEl = $('#api-books-grid');

    if (gridEl) {
        gridEl.innerHTML = pageBooks.length
            ? pageBooks.map(card).join('')
            : `<div class="col-12">
                 <p class="text-muted">KhÃ´ng tÃ¬m tháº¥y sÃ¡ch phÃ¹ há»£p.</p>
               </div>`;

        // Táº¡o vÃ¹ng phÃ¢n trang náº¿u chÆ°a cÃ³
        let pagination = $('#api-book-pagination');

        if (!pagination) {
            pagination = document.createElement('nav');
            pagination.id = 'api-book-pagination';
            pagination.setAttribute('aria-label', 'PhÃ¢n trang sÃ¡ch');
            gridEl.insertAdjacentElement('afterend', pagination);
        }

        // Táº¡o tá»‘i Ä‘a 5 sá»‘ trang
        let paginationHTML = '';

        if (totalPages > 1) {
            let startPage = Math.max(
                1,
                currentBookPage - Math.floor(MAX_VISIBLE_PAGES / 2)
            );

            let endPage = Math.min(
                totalPages,
                startPage + MAX_VISIBLE_PAGES - 1
            );

            startPage = Math.max(
                1,
                endPage - MAX_VISIBLE_PAGES + 1
            );

            const button = (page, label, disabled = false, active = false) => `
                <button
                    type="button"
                    class="btn ${active ? 'btn-primary' : 'btn-outline-primary'} mx-1"
                    data-page="${page}"
                    ${disabled ? 'disabled' : ''}
                    ${active ? 'aria-current="page"' : ''}
                    aria-label="${label}">
                    ${label}
                </button>
            `;

            paginationHTML += `
                <div class="d-flex justify-content-center align-items-center flex-wrap gap-1 mt-4 mb-4">
                    ${button(1, '&laquo;', currentBookPage === 1)}
                    ${button(
                        currentBookPage - 1,
                        '&lsaquo;',
                        currentBookPage === 1
                    )}
            `;

            for (let page = startPage; page <= endPage; page++) {
                paginationHTML += button(
                    page,
                    page,
                    false,
                    page === currentBookPage
                );
            }

            paginationHTML += `
                    ${button(
                        currentBookPage + 1,
                        '&rsaquo;',
                        currentBookPage === totalPages
                    )}
                    ${button(
                        totalPages,
                        '&raquo;',
                        currentBookPage === totalPages
                    )}
                </div>
            `;
        }

        pagination.innerHTML = paginationHTML;

        // Gáº¯n sá»± kiá»‡n cho cÃ¡c nÃºt phÃ¢n trang
        pagination.querySelectorAll('button[data-page]').forEach(btn => {
            btn.addEventListener('click', () => {
                const page = Number(btn.dataset.page);

                if (
                    page < 1 ||
                    page > totalPages ||
                    page === currentBookPage
                ) {
                    return;
                }

                currentBookPage = page;
                renderBookGrids();

                // Cuá»™n vá» Ä‘áº§u danh sÃ¡ch sÃ¡ch
                gridEl.scrollIntoView({
                    behavior: 'smooth',
                    block: 'start'
                });
            });
        });
    }

    // 6. Cáº­p nháº­t thÃ´ng bÃ¡o
    const status = $('#api-book-status');

    if (status) {
        if (totalBooks === 0) {
            status.textContent = 'KhÃ´ng tÃ¬m tháº¥y sÃ¡ch phÃ¹ há»£p.';
        } else {
            const from = startIndex + 1;
            const to = Math.min(startIndex + BOOKS_PER_PAGE, totalBooks);

            status.textContent =
                `Hiá»ƒn thá»‹ ${from}â€“${to} trong tá»•ng sá»‘ ${totalBooks} cuá»‘n sÃ¡ch.`;
        }
    }
}

function escapeCartText(value) {
    return String(value ?? '').replace(/[&<>"']/g, char => ({
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#39;'
    })[char]);
}

                Number(item.MaSach ?? item.maSach) !== id

            saveGuestCart(cart);
            await loadCart();
            updateCartBadge();
            notice('ÄÃ£ xÃ³a khá»i giá» hÃ ng');
        }

        return;
    }

    // TÄ‚NG / GIáº¢M Sá» LÆ¯á»¢NG
    const id = Number(changeButton.dataset.cartChange);
    const delta = Number(changeButton.dataset.delta);

    if (customerId != null) {
        await changeQty(id, delta);
    } else {
        let cart = getGuestCart();

        const item = cart.find(
            x => Number(x.MaSach ?? x.maSach) === id
        );

        if (item) {
            item.SoLuong = Number(item.SoLuong ?? item.soLuong ?? 1) + delta;

            if (item.SoLuong <= 0) {
                cart = cart.filter(
                    x => Number(x.MaSach ?? x.maSach) !== id
                );
            }
        }

        saveGuestCart(cart);
        await loadCart();
        updateCartBadge();
    }
});

async function syncGuestCartToDatabase() {
    const user = getUser();

    if (!user) return;

    const customerId = Number(
        user.maKhachHang ??
        user.MaKhachHang ??
        user.id ??
        user.Id
    );

    if (!customerId) {
        throw new Error('KhÃ´ng tÃ¬m tháº¥y mÃ£ khÃ¡ch hÃ ng.');
    }

    const guestCart = getGuestCart();

    if (guestCart.length === 0) return;

    for (const item of guestCart) {
        const bookId = Number(item.MaSach);
        const quantity = Math.max(1, Number(item.SoLuong || 1));

        // API hiá»‡n táº¡i thÃªm má»™t Ä‘Æ¡n vá»‹ sÃ¡ch má»—i láº§n gá»i.
        // Gá»i nhiá»u láº§n Ä‘á»ƒ chuyá»ƒn Ä‘á»§ sá»‘ lÆ°á»£ng tá»« giá» khÃ¡ch.
        for (let i = 0; i < quantity; i++) {
            await request(
                `/cart/customer/add?maKhachHang=${customerId}`,
                {
                    method: 'POST',
                    body: JSON.stringify({ MaSach: bookId })
                }
            );
        }
    }

    // Chá»‰ xÃ³a giá» táº¡m khi Ä‘á»“ng bá»™ thÃ nh cÃ´ng toÃ n bá»™
    sessionStorage.removeItem(GUEST_CART_KEY);

    updateGuestCartCount();

    // Táº£i giá» hÃ ng chÃ­nh thá»©c tá»« database
    if (typeof loadCart === 'function') {
        await loadCart();
    }

    notice('ÄÃ£ Ä‘á»“ng bá»™ giá» hÃ ng thÃ nh cÃ´ng.');
}


const GUEST_CART_KEY = 'apiq-guest-cart';

// Láº¥y giá» hÃ ng khÃ¡ch vÃ£ng lai tá»« sessionStorage
function getGuestCart() {
    try {
        return JSON.parse(
            sessionStorage.getItem(GUEST_CART_KEY) || '[]'
        );
    } catch {
        return [];
    }
}

// LÆ°u giá» hÃ ng khÃ¡ch vÃ£ng lai
function saveGuestCart(cart) {
    sessionStorage.setItem(
        GUEST_CART_KEY,
        JSON.stringify(cart)
    );
}

async function addGuestBook(book) {
    const id = Number(
        book.maSach ?? book.MaSach ?? book.id ?? book.Id
    );

    if (!id) {
        notice('KhÃ´ng xÃ¡c Ä‘á»‹nh Ä‘Æ°á»£c mÃ£ sÃ¡ch.');
        return;
    }

    const cart = getGuestCart();
    const existing = cart.find(
        item => Number(item.MaSach) === id
    );

    if (existing) {
        existing.SoLuong = Number(existing.SoLuong || 1) + 1;
        existing.HinhAnh = bookImage(book);
    } else {
        cart.push({
            MaSach: id,
            SoLuong: 1,
            TenSach: book.tenSach ?? book.TenSach ?? '',
            Gia: Number(book.gia ?? book.Gia ?? 0),
            HinhAnh: bookImage(book)
        });
    }

    saveGuestCart(cart);

    if (typeof renderGuestCart === 'function') {
        renderGuestCart();
    }

    notice('ÄÃ£ thÃªm sÃ¡ch vÃ o giá» hÃ ng.');
}

function bindBookViewMode() {
    const gridButton = document.querySelector('#book-grid-view');
    const listButton = document.querySelector('#book-list-view');
    const bookGrid = document.querySelector('#api-books-grid');

    if (!gridButton || !listButton || !bookGrid) return;

    gridButton.addEventListener('click', function (event) {
        event.preventDefault();

        bookGrid.classList.remove('book-list-view');
        gridButton.classList.add('active');
        listButton.classList.remove('active');
    });

    listButton.addEventListener('click', function (event) {
        event.preventDefault();

        bookGrid.classList.add('book-list-view');
        listButton.classList.add('active');
        gridButton.classList.remove('active');
    });
}



async function loadCart() {
    const user = getUser();

    console.log('[Giá» hÃ ng] NgÆ°á»i dÃ¹ng:', user);

    try {
        if (user && idOf(user)) {
            // ÄÃ£ Ä‘Äƒng nháº­p: láº¥y giá» hÃ ng tá»« database
            const customerId = idOf(user);

            const data = await request(
                `/cart/customer/${customerId}`
            );

            console.log('[Giá» hÃ ng] API tráº£ vá»:', data);

            const result = unwrap(data);

            // Há»— trá»£ API tráº£ vá» máº£ng trá»±c tiáº¿p hoáº·c bá»c trong data/result/items
            const items = Array.isArray(result)
                ? result
                : result?.items ?? result?.Items;

            state.cart = Array.isArray(items) ? items : [];
        } else {
            // ChÆ°a Ä‘Äƒng nháº­p: láº¥y giá» hÃ ng tá»« sessionStorage
            state.cart = getGuestCart();
        }
    } catch (error) {
        console.error('[Giá» hÃ ng] Lá»—i táº£i:', error);

        state.cart = [];

        notice(
            error.message || 'KhÃ´ng táº£i Ä‘Æ°á»£c giá» hÃ ng.',
            'error'
        );
    }

    renderCart();
    updateCartBadge();

    console.log('[Giá» hÃ ng] Dá»¯ liá»‡u cuá»‘i cÃ¹ng:', state.cart);

    return state.cart;

========== API\Extension\AdminOnlyAttribute.cs ==========
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;

namespace API.Extension
{
    public class AdminOnlyAttribute : Attribute, IAuthorizationFilter
    {
        public void OnAuthorization(AuthorizationFilterContext context)
        {
            var session = context.HttpContext.Session;

            var role = session.GetString(mySetting.Role);

            if (string.IsNullOrEmpty(role))
            {
                context.Result = new UnauthorizedObjectResult(new
                {
                    success = false,
                    message = "Bạn chưa đăng nhập!"
                });

                return;
            }

            if (role != "Admin")
            {
                context.Result = new ObjectResult(new
                {
                    success = false,
                    message = "Bạn không có quyền Admin!"
                })
                {
                    StatusCode = StatusCodes.Status403Forbidden
                };
            }
        }
    }
}
