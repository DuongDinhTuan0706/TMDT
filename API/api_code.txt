namespace API.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class HoaDonController : ControllerBase
    {
        private readonly IHoaDonServices _hoaDonServices;
        private readonly QlbanSachContext _context;

        public HoaDonController(
            IHoaDonServices hoaDonServices,
            QlbanSachContext context)
        {
            _hoaDonServices = hoaDonServices;
            _context = context;
        }


        // =====================================================
        // USER - LỊCH SỬ MUA HÀNG
        // =====================================================

        [HttpGet("lich-su")]
        public async Task<IActionResult>
            GetLichSuMuaHang()
        {
            var maKhachHang =
                HttpContext.Session
                    .GetInt32(mySetting.MaKhachHang);


            if (!maKhachHang.HasValue)
            {
                return Unauthorized(new
                {
                    message =
                        "Bạn chưa đăng nhập!"
                });
            }


            // Kiểm tra đã xác thực email chưa
            var khachHang =
                await _context.KhachHangs
                    .FirstOrDefaultAsync(x =>
                        x.MaKhachHang ==
                        maKhachHang.Value);


            if (khachHang == null)
            {
                return Unauthorized();
            }


            if (!khachHang.IsVerified)
            {
                return StatusCode(403, new
                {
                    message =
                        "Bạn cần xác thực email trước khi xem lịch sử mua hàng!"
                });
            }


            var result =
                await _hoaDonServices
                    .GetLichSuMuaHangAsync(
                        maKhachHang.Value);


            return Ok(result);
        }


        // =====================================================
        // USER - CHI TIẾT HÓA ĐƠN
        // =====================================================

                HttpContext.Session
                    .GetInt32(mySetting.MaKhachHang);

            {
                return Unauthorized(new
                {
                    message =
                        "Bạn chưa đăng nhập!"
                });
            }


            // Kiểm tra xác thực email
            var khachHang =
                await _context.KhachHangs
                    .FirstOrDefaultAsync(x =>

        public async Task<KhachHangResponseDTO> UpdateMyInfo(
    ThongTinKhachHangDTO dto)
        {
            var session = _httpContextAccessor.HttpContext?.Session;

            if (session == null)
                throw new Exception("Không tìm thấy Session!");

            var maKH = session.GetInt32(mySetting.MaKhachHang);

            if (!maKH.HasValue)
                throw new Exception("Bạn chưa đăng nhập!");

            var khachHang = await _context.KhachHangs
                .FirstOrDefaultAsync(x => x.MaKhachHang == maKH.Value);

            if (khachHang == null)
                throw new Exception("Không tìm thấy khách hàng!");

            if (!khachHang.IsVerified)
                throw new Exception(
                    "Bạn cần xác thực email trước khi chỉnh sửa thông tin!");

            if (string.IsNullOrWhiteSpace(dto.TenKhachHang))
                throw new Exception("Họ tên không được để trống!");

            if (string.IsNullOrWhiteSpace(dto.SoDienThoai))
                throw new Exception("Số điện thoại không được để trống!");

            if (string.IsNullOrWhiteSpace(dto.Email))
                throw new Exception("Email không được để trống!");

            var email = dto.Email.Trim();
            var soDienThoai = dto.SoDienThoai.Trim();

            // Kiểm tra email có thuộc khách hàng khác không
            var emailExists = await _context.KhachHangs.AnyAsync(x =>
                x.Email == email &&
                x.MaKhachHang != maKH.Value);

            if (emailExists)
                throw new Exception("Email đã được khách hàng khác sử dụng!");

            // Kiểm tra số điện thoại có thuộc khách hàng khác không
            var phoneExists = await _context.KhachHangs.AnyAsync(x =>
                x.SoDienThoai == soDienThoai &&
                x.MaKhachHang != maKH.Value);

            if (phoneExists)
                throw new Exception(
                    "Số điện thoại đã được khách hàng khác sử dụng!");

            // Cập nhật đúng khách hàng đang đăng nhập
            khachHang.TenKhachHang = dto.TenKhachHang.Trim();
            khachHang.SoDienThoai = soDienThoai;
            khachHang.DiaChi = string.IsNullOrWhiteSpace(dto.DiaChi)
                ? null
                : dto.DiaChi.Trim();

            // Email chỉ được giữ nguyên nếu không có yêu cầu đổi email.
            // Nếu cho phép đổi email, bỏ comment dòng dưới:
            // khachHang.Email = email;

            // Lưu vào SQL Server
            await _context.SaveChangesAsync();

            // Đồng bộ Session
            session.SetString(
                mySetting.TenKhachHang,
                khachHang.TenKhachHang ?? "");

            session.SetString(
                mySetting.SoDienThoai,
                khachHang.SoDienThoai ?? "");

            session.SetString(
                mySetting.Email,
                khachHang.Email ?? "");

            // Trả về dữ liệu vừa cập nhật
            return MapKhachHang(khachHang);
        }



        public async Task<bool> VerifyOTPAsync(
            int maKhachHang,
            string otp)
        {
            var khachHang =
                await _context.KhachHangs
                    .FirstOrDefaultAsync(x =>
                        x.MaKhachHang ==
                        maKhachHang);
            if (khachHang == null)
            {
                return false;
            }
            if (khachHang.IsVerified)
            {
                return true;
            }
            bool isValid =
                _otpService.VerifyOTP(
                    maKhachHang,
                    otp);
            if (!isValid)
            {
                return false;
            }
            khachHang.IsVerified = true;
            await _context.SaveChangesAsync();
            return true;
        }

                    TenSach = x.TenSach,
                    Gia = Convert.ToDouble(x.Gia),
                    AnhBia = x.AnhBia,

                        : null,

                        ? x.MaTacGiaNavigation.TenTacGia
                        : null,

                        ? x.MaTheLoaiNavigation.TenTheLoai
                        : null
                })

            return result;

        public async Task<IEnumerable<SachDTO>> GetListByConditionAsync(SearchDTO searchDto)
        {
            var query = _context.Saches.AsQueryable();
            if (!string.IsNullOrEmpty(searchDto.TenSach))
            {
                query = query.Where(x => x.TenSach.Contains(searchDto.TenSach));
            }
            if (searchDto.MaTheLoai != null)
            {
                query = query.Where(x => x.MaTheLoai == searchDto.MaTheLoai);
            }

            if (searchDto.MaTacGia != null)
            {
                query = query.Where(x =>x.MaTacGia == searchDto.MaTacGia);
            }
            if (searchDto.GiaMin.HasValue)
            {
                query = query.Where(x =>x.Gia >= (decimal)searchDto.GiaMin.Value);
            }
            if (searchDto.GiaMax.HasValue)
            {
                query = query.Where(x =>x.Gia <= (decimal)searchDto.GiaMax.Value);

                .Select(x => new SachDTO
                {
                    MaSach = x.MaSach,
